#--------------------------------------------------
# Subtree Sync Script
# Syncs subtree repositories into host repo
#--------------------------------------------------

# Only run on main, develop, or stage branches
if [[ "$SOURCE_BRANCH" != "main" && "$SOURCE_BRANCH" != "develop" && "$SOURCE_BRANCH" != "stage" && "$SOURCE_BRANCH" != "release/prod-familyins-1.0" ]]; then

  echo "   Current branch: $SOURCE_BRANCH"

echo "✓ Running on branch: $SOURCE_BRANCH"

# Install git-subtree

if ! curl -fsSL -o /usr/local/bin/git-subtree https://raw.githubusercontent.com/git/git/master/contrib/subtree/git-subtree.sh; then
  echo "❌ Failed to download git-subtree"
  exit 1
fi
chmod +x /usr/local/bin/git-subtree
echo "✓ git-subtree installed"

#--------------------------------------------------
# Configure git
#--------------------------------------------------
git config --global user.email "buildmaster-definity@economical.com"
git config --global user.name "buildmaster-definity"

# Configure rebase to always prefer our changes in conflicts
git config --global pull.rebase true
git config --global rebase.autoStash true

#--------------------------------------------------
# Configure authentication
#--------------------------------------------------
if [ -z "$BITBUCKET_WORKSPACE_ACCESS_TOKEN" ]; then
  echo "❌ ERROR: BITBUCKET_WORKSPACE_ACCESS_TOKEN not set"
  exit 1
fi

git config --global url."https://x-token-auth:$BITBUCKET_WORKSPACE_ACCESS_TOKEN@bitbucket.org/".insteadOf "https://bitbucket.org/"

#--------------------------------------------------
# Validate required variables
#--------------------------------------------------
if [ -z "$SOURCE_REPO" ]; then
  echo "❌ ERROR: SOURCE_REPO not set"
  exit 1
fi

#--------------------------------------------------
# Prepare repository state
#--------------------------------------------------
echo "Current git state:"
git status --short

# Clean working directory completely (CI/CD doesn't need stash)

echo "Cleaning working directory..."
git reset --hard HEAD
git clean -fd

echo "Fetching all branches from origin..."
git fetch origin

# Store the remote commit before checkout
REMOTE_COMMIT_BEFORE=$(git rev-parse FETCH_HEAD)
echo "Remote $SOURCE_BRANCH at: $(git rev-parse --short FETCH_HEAD)"

echo ""
echo "Checking out $SOURCE_BRANCH..."

# Always create/recreate branch from remote

echo "✓ Checked out $SOURCE_BRANCH at $(git rev-parse --short HEAD)"

# Subtree configuration
#--------------------------------------------------
SUBTREE_PREFIX="$SOURCE_REPO"
SUBTREE_REMOTE="https://bitbucket.org/economical/$SOURCE_REPO.git"

echo "Subtree: $SUBTREE_REMOTE"
echo "Prefix:  $SUBTREE_PREFIX"

#--------------------------------------------------
# Determine subtree branch to pull
#--------------------------------------------------
SUBTREE_BRANCH="$SOURCE_BRANCH"

echo ""
echo "Checking if subtree branch '$SUBTREE_BRANCH' exists..."
if ! git ls-remote --exit-code --heads "$SUBTREE_REMOTE" "$SUBTREE_BRANCH" > /dev/null 2>&1; then
  echo "⚠️  Subtree branch '$SUBTREE_BRANCH' does not exist"
  echo "   Falling back to 'main'"
  SUBTREE_BRANCH="main"

  # Verify main exists
  if ! git ls-remote --exit-code --heads "$SUBTREE_REMOTE" "$SUBTREE_BRANCH" > /dev/null 2>&1; then
    echo "❌ ERROR: Neither '$SOURCE_BRANCH' nor 'main' exists in subtree repo"
    exit 1
  fi
fi

#--------------------------------------------------
# Check if subtree was added with --squash
#--------------------------------------------------

if [ -d "$SUBTREE_PREFIX" ]; then
 
  #fetch full git history to detect squash for definity-aem-broker
  git fetch --unshallow 2>/dev/null || git fetch depth=214748647 2>/dev/null || true

  echo ""
  echo "Checking how subtree was originally added..."

  # Look for squash commits in the history
  if git log --all --grep="git-subtree-dir: $SUBTREE_PREFIX" --oneline 2>/dev/null | head -5 | grep -qi "squash"; then
    USES_SQUASH=true
    echo "✓ Subtree was added with --squash"
  else
    echo "✓ Subtree was added without --squash"
  fi
fi

#--------------------------------------------------
# Sync subtree
#--------------------------------------------------
echo ""
if [ ! -d "$SUBTREE_PREFIX" ]; then
  echo "📦 Subtree folder does not exist. Adding it..."
  git subtree add --prefix="$SUBTREE_PREFIX" "$SUBTREE_REMOTE" "$SUBTREE_BRANCH" --squash -m "Add $SOURCE_REPO subtree from $SUBTREE_BRANCH"
  echo "✓ Subtree added successfully"
else
  echo "📦 Subtree folder exists. Pulling latest changes..."

  # Fetch the latest changes
  echo "Fetching from $SUBTREE_REMOTE ($SUBTREE_BRANCH)..."
  git fetch "$SUBTREE_REMOTE" "$SUBTREE_BRANCH"

  # Get the current commit of the subtree remote
  REMOTE_COMMIT=$(git rev-parse FETCH_HEAD)
  echo "Remote subtree at: $REMOTE_COMMIT"

  # Pull with or without squash based on how it was originally added
  # Pull with or without squash based on how it was originally added
  if [ "$USES_SQUASH" = true ]; then
    echo "Pulling with --squash..."
    git subtree pull --prefix="$SUBTREE_PREFIX" "$SUBTREE_REMOTE" "$SUBTREE_BRANCH" --squash -m "Update $SOURCE_REPO subtree from $SUBTREE_BRANCH"
  else
    echo "Pulling without --squash..."
    git subtree pull --prefix="$SUBTREE_PREFIX" "$SUBTREE_REMOTE" "$SUBTREE_BRANCH" -m "Update $SOURCE_REPO subtree from $SUBTREE_BRANCH"
  fi

  echo "✓ Subtree updated successfully"
fi

#--------------------------------------------------
# Check if there are changes to push
#--------------------------------------------------
echo ""
echo "Checking for changes to push..."

# Fetch latest remote state
echo "Fetching latest remote state..."
git fetch origin "$SOURCE_BRANCH"

# Compare commits
LOCAL_COMMIT=$(git rev-parse HEAD)
REMOTE_COMMIT_AFTER=$(git rev-parse FETCH_HEAD)

echo "Commit comparison:"
echo "  Local:        $LOCAL_COMMIT"
echo "  Remote After: $REMOTE_COMMIT_AFTER"
echo "  Remote Before: $REMOTE_COMMIT_BEFORE"

# Check if already up to date
if [ "$LOCAL_COMMIT" = "$REMOTE_COMMIT_AFTER" ]; then
  echo "ℹ️  No changes to push - already up to date"
  exit 0
fi

# Check if we can fast-forward (remote hasn't changed)
if [ "$REMOTE_COMMIT_BEFORE" = "$REMOTE_COMMIT_AFTER" ]; then
  echo "✓ Remote unchanged - safe to push"
else
  echo "⚠️  Remote has changed during execution"
  echo "   Using force-with-lease strategy"
fi

#--------------------------------------------------
# Push changes with retry logic
#--------------------------------------------------
echo ""
echo "Pushing changes to origin/$SOURCE_BRANCH..."

# Try 1: Normal push (works if fast-forward)
if git push origin "HEAD:$SOURCE_BRANCH" 2>/dev/null; then
  echo "✓ Normal push successful"
elif git push --force-with-lease="$SOURCE_BRANCH:$REMOTE_COMMIT_BEFORE" origin "HEAD:$SOURCE_BRANCH" 2>/dev/null; then
  # Try 2: Force-with-lease using our initial fetch (prevents overwriting others' work)
  echo "✓ Force-with-lease push successful"
else
  # Try 3: Fetch again and retry with updated lease
  echo "⚠️  Push failed, fetching and retrying..."
  git fetch origin "$SOURCE_BRANCH"
  LATEST_REMOTE=$(git rev-parse FETCH_HEAD)

  if git push --force-with-lease="$SOURCE_BRANCH:$LATEST_REMOTE" origin "HEAD:$SOURCE_BRANCH"; then
    echo "✓ Retry with updated lease successful"
  else
    echo "❌ All push attempts failed"
    echo "This may indicate a race condition with another pipeline run"
    exit 1
  fi
fi

echo ""
echo "✅ Subtree sync for branch '$SOURCE_BRANCH' completed successfully"
echo "   Subtree: $SOURCE_REPO ($SUBTREE_BRANCH)"
echo "   Local:  $(git rev-parse --short HEAD)"

This repository contains the Bitbucket pipeline for the Digital CMS Platform. It is responsible for updating and managing the remote repos folder through git subtree concept.

## Pipeline Steps

The pipeline consists of two main steps:

2.  **Veracode Scan:** This step performs a security scan on the codebase using Veracode. It checks for vulnerabilities and will fail the pipeline if any high-severity issues are found.

The pipeline is configured to run only when the source branch is either `develop` or `main`. This ensures that only code from these branches is processed.

*   `definity-aem-broker`
*   `definity-aem`

*   The `develop` branch of the submodule will be pulled into the `develop` branch of this repository.
*   The `main` branch of the submodule will be pulled into the `main` branch of this repository.

**This pipeline should not be triggered manually.** It is designed to be triggered automatically by changes in the submodule repositories.

## One important change from your Bitbucket implementation

I would **not ask the coding agent to blindly translate your `update_submodule.sh` line-for-line**.

Your Bitbucket script has some things that I would simplify for GitHub.

For example, this part:

```bash
git checkout -B "$SOURCE_BRANCH" FETCH_HEAD
```

combined with:

```bash
git push --force-with-lease
```

is more aggressive than I'd want for the new GitHub PoC.

For GitHub, I'd aim for:

```text
Source repo
    │
    │ push/merge
    ▼
GitHub Actions
    │
    │ repository_dispatch
    │ SOURCE_REPO=project1
    │ SOURCE_BRANCH=develop
    ▼
Container workflow
    │
    ├── concurrency lock
    │
    ├── checkout develop
    │
    ├── git subtree pull
    │
    ├── mvn verify
    │
    └── git push origin develop
```

That gives you a much cleaner model.

### Also, I'd separate these two concepts

Your current Bitbucket terminology says **"submodule"**, but you're actually using **subtree**.

For the GitHub project I'd consistently call it:

> **source repository → container repository → Git subtree synchronization**

rather than:

> submodule update

This will make the architecture much easier to explain to others.
